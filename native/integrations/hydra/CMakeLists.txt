cmake_minimum_required(VERSION 3.26)
# Set before compiler detection, including the later Objective-C++ compiler.
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum supported macOS version")
endif()
project(hdMuTracer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
set(MUTRACER_BLENDER_ROOT "" CACHE PATH "Blender 5.2 application bundle or Windows directory")
if(MUTRACER_BLENDER_ROOT)
  # Do not import the matching development build's USD targets: their transitive
  # dependencies would load a second USD/Python/TBB into Blender's process.
  find_path(MUTRACER_USD_INCLUDE_DIR pxr/pxr.h
    HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include REQUIRED)
  find_path(MUTRACER_TBB_INCLUDE_DIR tbb/tbb.h
    HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include REQUIRED)
  file(READ "${MUTRACER_USD_INCLUDE_DIR}/pxr/pxr.h" usd_header)
  if(NOT usd_header MATCHES "#define PXR_INTERNAL_NS pxrBlender_v26_03__pxrReserved__"
      OR NOT usd_header MATCHES "#define PXR_VERSION 2603"
      OR NOT usd_header MATCHES "#define PXR_PYTHON_SUPPORT_ENABLED")
    message(FATAL_ERROR "Blender needs Python-enabled USD 26.03 headers with pxrBlender_v26_03 namespace")
  endif()
  file(READ "${MUTRACER_USD_INCLUDE_DIR}/pxr/base/arch/attributes.h" arch_header)
  if((APPLE AND NOT arch_header MATCHES "__DATA,pxbctor")
      OR (WIN32 AND NOT arch_header MATCHES "\\.pxbctor"))
    message(FATAL_ERROR "Blender's USD headers must use its pxbctor/pxbdtor registration sections")
  endif()
  if(WIN32)
    set(blender_lib "${MUTRACER_BLENDER_ROOT}/blender.shared")
  else()
    set(blender_lib "${MUTRACER_BLENDER_ROOT}/Contents/Resources/lib")
  endif()
  find_path(MUTRACER_PYTHON_INCLUDE_DIR Python.h
    HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include/python3.13 include REQUIRED)
  set(blender_python "${MUTRACER_PYTHON_INCLUDE_DIR}")
  set(PXR_INCLUDE_DIRS "${MUTRACER_USD_INCLUDE_DIR}" "${MUTRACER_TBB_INCLUDE_DIR}" "${blender_python}")
  if(WIN32)
    find_library(MUTRACER_BLENDER_USD_LIBRARY usd_ms
      HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES lib REQUIRED NO_DEFAULT_PATH)
    find_library(MUTRACER_BLENDER_TBB_LIBRARY tbb12
      HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES lib REQUIRED NO_DEFAULT_PATH)
    find_library(MUTRACER_BLENDER_PYTHON_LIBRARY python313
      HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES libs lib REQUIRED NO_DEFAULT_PATH)
    if(NOT EXISTS "${blender_lib}/usd_ms.dll")
      message(FATAL_ERROR "Expected Blender 5.2 with blender.shared/usd_ms.dll")
    endif()
    set(mutracer_usd_libraries "${MUTRACER_BLENDER_USD_LIBRARY}"
      "${MUTRACER_BLENDER_TBB_LIBRARY}" "${MUTRACER_BLENDER_PYTHON_LIBRARY}")
  else()
    if(NOT EXISTS "${blender_lib}/libusd_ms.dylib")
      message(FATAL_ERROR "Expected a Blender 5.2 macOS bundle with libusd_ms.dylib")
    endif()
    set(mutracer_usd_libraries "${blender_lib}/libusd_ms.dylib" "${blender_lib}/libtbb.dylib")
  endif()
else()
  find_package(pxr CONFIG REQUIRED)
  set(mutracer_usd_libraries hd hdx hgi glf garch usdImagingGL usdImaging usdLux usdShade usdGeom usd sdf sdr js tf gf vt hio hf cameraUtil)
  if(APPLE)
    list(APPEND mutracer_usd_libraries hgiMetal)
  endif()
endif()
get_filename_component(MUTRACER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MUTRACER_LIBRARY_DIR "${MUTRACER_ROOT}/.native-build/scene-core/cargo-target/release" CACHE PATH "Built SDK directory")
if(WIN32)
  find_library(MUTRACER_LIBRARY NAMES mutracer.dll PATHS "${MUTRACER_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
  find_file(MUTRACER_RUNTIME mutracer.dll PATHS "${MUTRACER_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
else()
  find_library(MUTRACER_LIBRARY mutracer PATHS "${MUTRACER_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
  set(MUTRACER_RUNTIME "${MUTRACER_LIBRARY}")
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(mutracer_gpu_adapter gpuMetal.mm)
  set_source_files_properties(gpuMetal.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
else()
  set(mutracer_gpu_adapter gpuFallback.cpp)
endif()
add_library(hdMuTracer SHARED delegate.cpp scene.cpp material.cpp materialGraph.cpp gpu.cpp ${mutracer_gpu_adapter})
if(MSVC)
  target_compile_options(hdMuTracer PUBLIC /utf-8 /bigobj)
  target_compile_definitions(hdMuTracer PUBLIC NOMINMAX _USE_MATH_DEFINES)
  set_property(TARGET hdMuTracer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  set_property(TARGET hdMuTracer PROPERTY WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
target_include_directories(hdMuTracer PUBLIC ${PXR_INCLUDE_DIRS} "${MUTRACER_ROOT}/native/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(hdMuTracer PUBLIC ${mutracer_usd_libraries} "${MUTRACER_LIBRARY}")
if(APPLE)
  target_link_libraries(hdMuTracer PUBLIC "-framework Metal" "-framework Foundation")
endif()
set_target_properties(hdMuTracer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugin/hdMuTracer")
if(WIN32)
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_property(TARGET hdMuTracer PROPERTY RUNTIME_OUTPUT_DIRECTORY_${configuration}
      "${CMAKE_CURRENT_BINARY_DIR}/plugin/hdMuTracer")
  endforeach()
endif()
set_target_properties(hdMuTracer PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugin/hdMuTracer" BUILD_RPATH "${MUTRACER_LIBRARY_DIR};${PXR_LIBRARY_DIRS}" INSTALL_RPATH "${MUTRACER_LIBRARY_DIR};${PXR_LIBRARY_DIRS}")
configure_file(resources/plugInfo.json.in "${CMAKE_CURRENT_BINARY_DIR}/plugin/hdMuTracer/resources/plugInfo.json" @ONLY)
if(MUTRACER_BLENDER_ROOT AND APPLE)
  # VtValue's inline converters reference Blender's statically linked Python.
  # Keep USD symbols link-checked; only these interpreter symbols are deferred.
  foreach(symbol _PyBool_FromLong _PyFloat_FromDouble _PyLong_FromLong
      _PyUnicode_FromStringAndSize __Py_Dealloc __Py_NoneStruct)
    target_link_options(hdMuTracer PRIVATE "LINKER:-U,${symbol}")
  endforeach()
  set_target_properties(hdMuTracer PROPERTIES
    BUILD_WITH_INSTALL_RPATH ON
    INSTALL_RPATH "@loader_path;@executable_path/../Resources/lib")
  get_filename_component(mutracer_library_name "${MUTRACER_LIBRARY}" NAME)
  add_custom_command(TARGET hdMuTracer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MUTRACER_LIBRARY}"
      "$<TARGET_FILE_DIR:hdMuTracer>/${mutracer_library_name}")
endif()
if(MUTRACER_BLENDER_ROOT)
  install(TARGETS hdMuTracer LIBRARY DESTINATION hdMuTracer RUNTIME DESTINATION hdMuTracer)
  install(FILES "${MUTRACER_RUNTIME}" DESTINATION hdMuTracer)
  if(WIN32)
    file(GLOB mutracer_runtime_dlls "${MUTRACER_LIBRARY_DIR}/*.dll")
    install(FILES ${mutracer_runtime_dlls} DESTINATION hdMuTracer)
    set(MUTRACER_SDK_LICENSE_DIR "${MUTRACER_LIBRARY_DIR}/../licenses" CACHE PATH "Precompiled SDK licenses")
    if(NOT IS_DIRECTORY "${MUTRACER_SDK_LICENSE_DIR}")
      message(FATAL_ERROR "Windows add-on packaging requires the precompiled SDK license directory")
    endif()
    install(DIRECTORY "${MUTRACER_SDK_LICENSE_DIR}/" DESTINATION hdMuTracer/licenses)
  endif()
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/plugin/hdMuTracer/resources/plugInfo.json"
    DESTINATION hdMuTracer/resources)
else()
  enable_testing()
  add_executable(hydraContracts tests/contracts.cpp)
  target_link_libraries(hydraContracts PRIVATE hdMuTracer)
  target_compile_definitions(hydraContracts PRIVATE MT_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
  add_test(NAME hydraContracts COMMAND hydraContracts)
  if(APPLE)
    set(mutracer_slow_host tests/slowHost.mm)
    set_source_files_properties(tests/slowHost.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  else()
    set(mutracer_slow_host tests/slowHost.cpp)
  endif()
  add_executable(hydraTrace tests/trace.cpp ${mutracer_slow_host})
  target_link_libraries(hydraTrace PRIVATE hdMuTracer)
  add_executable(hydraGpuContracts tests/conversion.cpp)
  target_link_libraries(hydraGpuContracts PRIVATE hdMuTracer)
  add_test(NAME hydraGpuContracts COMMAND hydraGpuContracts)
  if(WIN32)
    # HgiGL emits its shader-int64 extension with the bindless-buffer feature.
    # This process-local test option preserves the integer IEEE conversion used
    # by Metal; adapters without the required GL extensions report an error.
    set_tests_properties(hydraGpuContracts PROPERTIES
      ENVIRONMENT "HGIGL_ENABLE_BINDLESS_BUFFER=1")
  endif()
endif()
